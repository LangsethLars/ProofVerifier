# To learn the theory behind finite state machines, regular expressions,
# grammar and parsers, I made a script language and a Parser project.
# This can be used to define new languages, tokens and LL(1) grammar,
# and generate the code for lexing and parsing this new language.

# This file will define a language inspired by Lean, but smaller with
# less syntactical sugar. Just a bare minimum to be able to verify
# proofs in mathematics. I want to understand how to make a theorem
# prover from scratch and write it in simple C++.

# Tokens that will be ignored
# ===========================

Ignore COMMENT = '#' [\01-\FF~\0A\0D]*;
Ignore SPACE   = [\09\20]+;
Ignore NL      = '\0A' | '\0D' | "\0D\0A" | "\0A\0D";

# Identifiers, always starts with lower case letter
# =================================================

Token IDENT    = [a-z] [_0-9A-Za-z]*;
Token NUMBER   = ('0' | [1-9][0-9]*) '-'?;

# Keywords, never starts with lower case letter
# =============================================

Token _Int              = "Int";

Token _Int_Add          = "Int.Add";
Token _Int_Sub          = "Int.Sub";
Token _Int_Mul          = "Int.Mul";
Token _Int_Neg          = "Int.Neg";

Token _Prop             = "Prop";

Token _Prop_True        = "Prop.True";
Token _Prop_False       = "Prop.False";

Token _Prop_Not         = "Prop.Not";
Token _Prop_And         = "Prop.And";
Token _Prop_Or          = "Prop.Or";
Token _Prop_Implies     = "Prop.Implies";
Token _Prop_Iff         = "Prop.Iff";

Token _Prop_AndLeft     = "Prop.AndLeft";
Token _Prop_AndRight    = "Prop.AndRight";
Token _Prop_AndIntro    = "Prop.AndIntro";
Token _Prop_OrLeft      = "Prop.OrLeft";
Token _Prop_OrRight     = "Prop.OrRight";
Token _Prop_FalseElim   = "Prop.FalseElim";

Token _Prop_Int_Eq      = "Prop.Int.Eq";    # Int.Eq(a b) returns Prop

Token _Theorem          = "Theorem";
Token _Proof            = "Proof";
Token _End              = "End";
Token _Assume           = "Assume";
Token _Have             = "Have";
Token _Exact            = "Exact";
Token _Show             = "Show";
Token _Cases            = "Cases";
Token _Case             = "Case";
Token _Refl             = "Refl";
Token _Trivial          = "Trivial";

# Wait with the following
#========================

Token _By           = "By";
Token _Rewrite      = "Rewrite";
Token _Symm         = "Symm";
Token _Trans        = "Trans";
Token _Reduce       = "Reduce";
Token _Def          = "Def";
Token _ForAll       = "ForAll";
Token _Exists       = "Exists";
Token _Induction    = "Induction";
Token _Fun          = "Fun";
Token _Pi           = "Pi";
Token _Type         = "Type";
Token _Nat          = "Nat";

# Symbols
# =======

Token EQUAL         = '=';
Token OPEN_PAR      = '(';
Token CLOSE_PAR     = ')';
Token COLON         = ':';
Token COLONEQUAL    = ":=";
Token COMMA         = ',';
Token DOT           = '.';
Token SLIM_ARROW    = "->";
Token FAT_ARROW     = "=>";
Token SEMICOLON     = ';';

# Grammar written in LL(1)
# ========================

# To avoid name collitions, I use ('<' alphanum* '>') for symbols, while
# tokens are (alphanum*). General tokens are UPPERCASE while fixed keywords
# are prefixed with _ followed by the keyword (if legal).
# That makes it easier to read as well.

# <Start> is the main entry point. An invisible temporary rule
# with a single production will call <Start> like this:
# Rule <> = <Start> END_OF_FILE;

Skip <Start>
    = <Item>;

Skip <Item>
    = <Theorem> <Item>
    | ;

Rule <Theorem>
    = _Theorem IDENT <ParamList> COLON <PropExpr> <ProofBlock>;

Rule <ParamList>
    = <Param> <ParamList>
    | ;

Rule <Param>
    = OPEN_PAR IDENT COLON <TypeExpr> CLOSE_PAR;

Skip <TypeExpr>
    = _Int
    | _Prop;

Rule <PropExpr>
    = IDENT
    | _Prop_True
    | _Prop_False
    | _Prop_Not     OPEN_PAR <PropExpr> CLOSE_PAR
    | _Prop_And     OPEN_PAR <PropExpr> <PropExpr> CLOSE_PAR
    | _Prop_Or      OPEN_PAR <PropExpr> <PropExpr> CLOSE_PAR
    | _Prop_Implies OPEN_PAR <PropExpr> <PropExpr> CLOSE_PAR
    | _Prop_Iff     OPEN_PAR <PropExpr> <PropExpr> CLOSE_PAR
    | _Prop_Int_Eq  OPEN_PAR <IntExpr>  <IntExpr>  CLOSE_PAR;

Rule <IntExpr>
    = IDENT
    | NUMBER
    | _Int_Neg  OPEN_PAR <IntExpr>            CLOSE_PAR
    | _Int_Add  OPEN_PAR <IntExpr> <IntExpr>  CLOSE_PAR
    | _Int_Sub  OPEN_PAR <IntExpr> <IntExpr>  CLOSE_PAR
    | _Int_Mul  OPEN_PAR <IntExpr> <IntExpr>  CLOSE_PAR;

Rule <ProofBlock>
    = _Proof <ProofStepList> _End;

Skip <ProofStepList>
    = <ProofStep> <ProofStepList>
    | ;

Skip <ProofStep>
    = <AssumeStep>
    | <HaveStep>
    | <ExactStep>
    | <ShowStep>
    | <CasesStep>
    | <ReflStep>
    | <TrivialStep>;

Rule <AssumeStep>
    = _Assume IDENT COLON <PropExpr>;

Rule <HaveStep>
    = _Have IDENT COLON <PropExpr> COLONEQUAL <ProofExpr>;

Rule <ExactStep>
    = _Exact <ProofExpr>;

Rule <ShowStep>
    = _Show <PropExpr>;

Rule <ReflStep>
    = _Refl;

Rule <TrivialStep>
    = _Trivial;

Rule <ProofExpr>
    = IDENT
    | _Refl
    | _Trivial
    | _Prop_FalseElim OPEN_PAR <PropExpr>  <ProofExpr> CLOSE_PAR
    | _Prop_AndLeft   OPEN_PAR <ProofExpr>             CLOSE_PAR
    | _Prop_AndRight  OPEN_PAR <ProofExpr>             CLOSE_PAR
    | _Prop_AndIntro  OPEN_PAR <ProofExpr> <ProofExpr> CLOSE_PAR
    | _Prop_OrLeft    OPEN_PAR <PropExpr>  <ProofExpr> CLOSE_PAR
    | _Prop_OrRight   OPEN_PAR <PropExpr>  <ProofExpr> CLOSE_PAR;

# Proof checking model
# ====================
#
# The proof checker maintains:
#
# - A context of variables:
#     x : Int
#     p : Prop
#
# - A context of proof assumptions:
#     h : p
#     hpq : Prop.And(p q)
#
# - A current goal:
#     goal : PropExpr
#
# Each ProofStep either:
# - modifies the current goal and context, or
# - adds a proof assumption, or
# - closes the current goal.
#
# See Examples.txt for implemented steps.
